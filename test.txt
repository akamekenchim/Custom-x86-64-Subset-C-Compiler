// =========================================================================
// HÀM BỔ TRỢ 1: Phép toán cộng / trừ cơ bản
// =========================================================================
int add(int a, int b) {
    return a + b;
}

int sub(int a, int b) {
    return a - b;
}

// =========================================================================
// HÀM BỔ TRỢ 2: Đệ quy đôi (Dual Recursion)
// =========================================================================
int fibonacci(int n) {
    if (n <= 0) {
        return 0;
    }
    if (n == 1) {
        return 1;
    }
    return fibonacci(n - 1) + fibonacci(n - 2);
}

// =========================================================================
// HÀM BỔ TRỢ 3: Thao tác Mảng qua Con trỏ (Pointer Offset & Pass-by-ref)
// =========================================================================
int mutate_array(int *ptr, int size) {
    int i;
    for (i = 0; i < size; i = i + 1) {
        // Gán ptr[i] = -ptr[i] * 2 thông qua giải tham chiếu *(ptr + i)
        *(ptr + i) = -(*(ptr + i)) * 2;
    }
    return 0;
}

// =========================================================================
// HÀM MAIN KIỂM THỬ MỌI GIỚI HẠN
// =========================================================================
int main() {
    int pass_count;

    // Biến Test 1: Unary Minus & Precedence
    int u_a;
    int u_b;
    int u_res;

    // Biến Test 2: Pointer Arithmetic
    int arr[5];
    int *p;
    int p_sum;

    // Biến Test 3: Dual Recursion & Unary Call
    int rec_res;

    // Biến Test 4: Control Flow & Bitwise
    int flow_i;
    int flow_sum;

    // Biến Test 5: Nested Function Calls
    int nest_res;

    pass_count = 0;

    printf("==================================================\n");
    printf("   COMPILER LIMIT & STRESS TEST SUITE (EXTREME)  \n");
    printf("==================================================\n");

    // ---------------------------------------------------------------------
    // TEST 1: Unary Minus & Complex Precedence
    // Công thức: u_res = -(-5) * -10 + (-20 / 4)
    // Tính toán: 5 * -10 + (-5) = -50 - 5 = -55
    // ---------------------------------------------------------------------
    printf("TEST 1: Unary Minus & Precedence Logic\n");
    u_a = -5;
    u_b = -(-10); // u_b = 10
    u_res = -u_a * -u_b + (-20 / 4);

    if (u_res == -55) {
        printf("-> TEST 1: PASS\n");
        pass_count = pass_count + 1;
    } else {
        printf("-> TEST 1: FAIL (Expected -55)\n");
    }

    // ---------------------------------------------------------------------
    // TEST 2: Pointer Arithmetic & Array Decay
    // Khởi tạo: [10, -20, 30, -40, 50]
    // Sau mutate: [-20, 40, -60, 80, -100]
    // p_sum = arr[1] + arr[3] = 40 + 80 = 120
    // ---------------------------------------------------------------------
    printf("TEST 2: Pointer Arithmetic & Array Mutation\n");
    arr[0] = 10;
    arr[1] = -20;
    arr[2] = 30;
    arr[3] = -40;
    arr[4] = 50;

    p = arr;
    mutate_array(p, 5);

    p_sum = *(p + 1) + *(p + 3); // 40 + 80

    if (p_sum == 120) {
        printf("-> TEST 2: PASS\n");
        pass_count = pass_count + 1;
    } else {
        printf("-> TEST 2: FAIL (Expected 120)\n");
    }

    // ---------------------------------------------------------------------
    // TEST 3: Dual Recursion Mixed with Unary Operations
    // Tính toán: fib(7) + -fib(5) = 13 + -5 = 8
    // ---------------------------------------------------------------------
    printf("TEST 3: Dual Recursion & ABI Register Protection\n");
    rec_res = fibonacci(7) + -fibonacci(5);

    if (rec_res == 8) {
        printf("-> TEST 3: PASS\n");
        pass_count = pass_count + 1;
    } else {
        printf("-> TEST 3: FAIL (Expected 8)\n");
    }

    // ---------------------------------------------------------------------
    // TEST 4: Flow Control (For + While + Break + Continue + Bitwise)
    // Vòng lặp tính tổng các số chẵn trong khoảng 1..10, bỏ qua 6, ngắt khi >= 10
    // Các số chẵn được chọn: 2, 4, 8 -> Tổng = 14
    // ---------------------------------------------------------------------
    printf("TEST 4: Complex Loop Control & Bitwise Flags\n");
    flow_sum = 0;
    flow_i = 0;

    while (flow_i < 10) {
        flow_i = flow_i + 1;

        // Bỏ qua số lẻ dùng Bitwise AND (& 1)
        if ((flow_i & 1) != 0) {
            continue;
        }

        if (flow_i == 6) {
            continue; // Bỏ qua 6
        }

        if (flow_i >= 10) {
            break; // Ngắt vòng lặp khi chạm 10
        }

        flow_sum = flow_sum + flow_i;
    }

    if (flow_sum == 14) {
        printf("-> TEST 4: PASS\n");
        pass_count = pass_count + 1;
    } else {
        printf("-> TEST 4: FAIL (Expected 14)\n");
    }

    // ---------------------------------------------------------------------
    // TEST 5: Deeply Nested Function Expressions
    // Tính toán: add( add(-10, 20), sub(5, -15) )
    // -> add( 10, 20 ) = 30
    // ---------------------------------------------------------------------
    printf("TEST 5: Deeply Nested Function Arguments\n");
    nest_res = add(add(-10, 20), sub(5, -15));

    if (nest_res == 30) {
        printf("-> TEST 5: PASS\n");
        pass_count = pass_count + 1;
    } else {
        printf("-> TEST 5: FAIL (Expected 30)\n");
    }

    // ---------------------------------------------------------------------
    // TỔNG KẾT BÀI TEST
    // ---------------------------------------------------------------------
    printf("==================================================\n");
    printf("TOTAL PASSED TESTS (OUT OF 5):\n");
    printf(pass_count);
    printf("\n");

    if (pass_count == 5) {
        printf(">>> CONGRATULATIONS! ALL LIMIT TESTS PASSED! <<<\n");
    } else {
        printf(">>> STRESS TEST DETECTED BUGS IN COMPILER! <<<\n");
    }
    printf("==================================================\n");

    return 0;
}
